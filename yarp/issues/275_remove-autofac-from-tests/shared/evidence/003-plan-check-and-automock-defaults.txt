// Question: what do AutoMock.GetLoose() mocks return for unscripted members, vs new Mock<T>()?
using Autofac.Extras.Moq;
using Moq;

public interface IFeatures { object Get(); }
public interface IServer { IFeatures Features { get; } }
public sealed class Concrete { public IServer Server { get; } public Concrete(IServer s) { Server = s; } }

public static class Program
{
    public static void Main()
    {
        using var am = AutoMock.GetLoose();
        var server = am.Mock<IServer>();
        Console.WriteLine($"AutoMock: Behavior={server.Behavior} DefaultValue={server.DefaultValue}");
        Console.WriteLine($"AutoMock: server.Features is null? {server.Object.Features is null}");
        var c = am.Create<Concrete>();
        Console.WriteLine($"AutoMock: Create<sealed Concrete> gave a real object? {c.GetType() == typeof(Concrete)}; its IServer is the shared mock? {ReferenceEquals(c.Server, server.Object)}");

        var plain = new Mock<IServer>();
        Console.WriteLine($"new Mock: Behavior={plain.Behavior} DefaultValue={plain.DefaultValue}");
        Console.WriteLine($"new Mock: server.Features is null? {plain.Object.Features is null}");
    }
}
